import re

def normalize_phone(s):
    """
    把任意格式的中国手机号归一化成 11 位纯数字字符串。
    例如 "138 0013 8000"、"138-0013-8000"、"+86 138 0013 8000" 都变成 "13800138000"。
    """
    # 1. 去掉所有非数字字符（空格、连字符、加号、括号等）
    digits = re.sub(r'\D', '', s)

    # 2. 处理开头的中国区号 86（对应 +86 86138... 之类，去重后多出的 86）
    #    若为 "+86 138..."，digits 为 "8613800138000"（13位），去掉开头的 86
    if digits.startswith('86') and len(digits) == 13:
        digits = digits[2:]

    # 3. 校验长度：合规的中国手机号应为 11 位
    if len(digits) != 11:
        raise ValueError(f"非法的中国手机号格式: {s!r}")

    return digits
